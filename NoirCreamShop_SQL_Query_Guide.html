NOIR & CREAM CLOTHING SHOP

คู่มือคำสั่ง SQL ที่จำเป็นสำหรับการค้นหาข้อมูล

รวมคำสั่ง SQL พร้อมใช้งานสำหรับค้นหา กรอง วิเคราะห์ และตรวจสอบข้อมูลตามโครงสร้าง ER Diagram และ Data Dictionary

💾 Database: [NoirCreamShop]
💻 Target: Microsoft SQL Server (SSMS) & SQLite
📑 รองรับ: 16 ตาราง + 2 Views (Orders, Order_Detail)
⚡ ระบบ Realtime 2-Way Sync
💡 ข้อแนะนำในการรันคำสั่งใน SSMS:
1. ก่อนรันคำสั่งทุกครั้ง ตรวจสอบว่าเลือก Database เป็น NoirCreamShop (หรือใส่คำสั่ง USE [NoirCreamShop]; ไว้บรรทัดแรก)
2. ข้อความภาษาไทยใน SQL Server ให้ใส่ตัวอักษร N นำหน้าเสมอ เช่น N'เสื้อยืด', N'โอนผ่านธนาคาร' เพื่อป้องกันภาษาไทยกลายเป็นเครื่องหมายคำถาม (?)
3. ตารางคำสั่งซื้อชื่อ order เป็น Keyword สงวนของ SQL ให้เขียนในรูปแบบ dbo.[order] หรือ "order" เสมอ
หมวด 1 คำสั่งค้นหาข้อมูลสินค้า สต็อก และไซส์ (Product & Variants)
1.1 ดูรายการสินค้าทั้งหมด พร้อมสต็อกรวมและราคาเริ่มต้น SELECT ALL
แสดงข้อมูลสินค้า หมวดหมู่ สถานะ สต็อกรวม และราคาของสินค้าทุกรายการในร้าน
SELECT product_id, name, category, price, stock, status, is_featured, created_at
FROM dbo.product
ORDER BY product_id DESC;
1.2 ค้นหาสินค้าตามหมวดหมู่เฉพาะ (เสื้อยืด, เสื้อโปโล, เสื้อแขนยาว, กางเกง) FILTER
ค้นหาเฉพาะสินค้าที่วางขาย (status = 'active') ในหมวดหมู่ที่ต้องการ
SELECT product_id, name, price, stock, description
FROM dbo.product
WHERE category = N'เสื้อยืด' AND status = 'active'
ORDER BY price ASC;
1.3 ค้นหาสินค้าด้วยคำค้น (ชื่อ หรือ รายละเอียด) LIKE SEARCH
ค้นหาคำที่ตรงกับชื่อหรือรายละเอียดสินค้า (เช่น ค้นหาคำว่า "Nike" หรือ "Cotton")
SELECT product_id, name, category, price, stock
FROM dbo.product
WHERE name LIKE N'%Nike%' OR description LIKE N'%Nike%';
1.4 ตรวจสอบสินค้าที่ใกล้หมดสต็อก (สต็อกเหลือน้อยกว่าหรือเท่ากับ 5 ชิ้น) ALERT / STOCK
ดูไซส์และสีของสินค้าแต่ละชิ้นที่สต็อกใกล้หมด เพื่อเตรียมสั่งผลิตหรือนำเข้าเพิ่ม
SELECT p.product_id, p.name, v.variant_id, v.color, v.size, v.stock, v.sku
FROM dbo.product_variant v
JOIN dbo.product p ON p.product_id = v.product_id
WHERE v.stock <= 5 AND p.status = 'active'
ORDER BY v.stock ASC;
1.5 ดูรูปแบบสินค้าทั้งหมด (JOIN สินค้า + ไซส์ + สี + ราคา + SKU) INNER JOIN
รวมตาราง product และ product_variant เพื่อดูรายละเอียดครบทุกไซส์ทุกสี
SELECT 
    p.product_id,
    p.name AS product_name,
    p.category,
    v.variant_id,
    v.size,
    v.color,
    v.price,
    v.stock,
    v.sku
FROM dbo.product p
JOIN dbo.product_variant v ON p.product_id = v.product_id
ORDER BY p.product_id, v.size;
หมวด 2 คำสั่งค้นหาข้อมูลคำสั่งซื้อและการชำระเงิน (Orders & Payments)
2.1 ดูรายการคำสั่งซื้อทั้งหมด เรียงจากล่าสุดไปเก่าสุด ORDERS LIST
แสดงรหัสคำสั่งซื้อ วันที่ ยอดรวม ส่วนลด ค่าส่ง และสถานะออเดอร์
SELECT 
    order_id,
    customer_id,
    order_date,
    subtotal,
    discount,
    shipping_fee,
    total_price,
    status,
    ship_name,
    ship_phone
FROM dbo.[order]
ORDER BY order_id DESC;
2.2 ค้นหาคำสั่งซื้อตามสถานะ (เช่น 'paid' ชำระแล้ว หรือ 'pending' รอดำเนินการ) STATUS FILTER
กรองดูเฉพาะออเดอร์ที่ชำระเงินแล้ว เพื่อเตรียมแพ็คสินค้าและจัดส่ง
SELECT order_id, customer_id, order_date, total_price, status, ship_name, ship_address
FROM dbo.[order]
WHERE status = 'paid'
ORDER BY order_date ASC;
2.3 ดูรายละเอียดคำสั่งซื้อแบบครบวงจร (JOIN ออเดอร์ + ลูกค้า + ชำระเงิน + พัสดุ) MULTI-JOIN
แสดงข้อมูลครบทุกด้าน: ใครสั่ง, ซื้ออะไร, จ่ายทางไหน, และส่งพัสดุเลขอะไร
SELECT 
    o.order_id,
    o.order_date,
    c.first_name + ' ' + c.last_name AS customer_name,
    c.phone,
    o.total_price,
    o.status AS order_status,
    pay.method AS payment_method,
    pay.status AS payment_status,
    ship.carrier,
    ship.tracking_no,
    ship.status AS shipping_status
FROM dbo.[order] o
JOIN dbo.customer c ON c.customer_id = o.customer_id
LEFT JOIN dbo.payment pay ON pay.order_id = o.order_id
LEFT JOIN dbo.shipment ship ON ship.order_id = o.order_id
ORDER BY o.order_id DESC;
2.4 ดูรายการสินค้าในคำสั่งซื้อแต่ละใบ (Order Items Breakdown) ORDER DETAILS
เจาะลึกออเดอร์ (เช่น Order_ID = 5) ว่าซื้อเสื้อตัวไหน ไซส์อะไร จำนวนกี่ตัว ราคาเท่าไร
SELECT 
    oi.order_id,
    p.name AS product_name,
    v.color,
    v.size,
    oi.quantity,
    oi.unit_price,
    oi.total_price
FROM dbo.order_item oi
JOIN dbo.product_variant v ON v.variant_id = oi.variant_id
JOIN dbo.product p ON p.product_id = v.product_id
WHERE oi.order_id = 5; -- เปลี่ยนหมายเลข order_id ตามต้องการ
หมวด 3 คำสั่งค้นหาข้อมูลลูกค้าและที่อยู่จัดส่ง (Customer & Addresses)
3.1 ดูรายชื่อลูกค้าทั้งหมด พร้อมยอดซื้อสะสมและจำนวนออเดอร์ CUSTOMER STATS
สรุปลูกค้าแต่ละคนว่าเคยสั่งซื้อกี่ครั้ง และมียอดสั่งซื้อสะสมรวมเท่าไร
SELECT 
    c.customer_id,
    c.first_name + ' ' + c.last_name AS customer_name,
    c.email,
    c.phone,
    COUNT(o.order_id) AS total_orders,
    COALESCE(SUM(o.total_price), 0) AS total_spent,
    c.created_at
FROM dbo.customer c
LEFT JOIN dbo.[order] o ON o.customer_id = c.customer_id
GROUP BY c.customer_id, c.first_name, c.last_name, c.email, c.phone, c.created_at
ORDER BY total_spent DESC;
3.2 ค้นหาลูกค้าด้วยอีเมล หรือ เบอร์โทรศัพท์ CUSTOMER LOOKUP
ค้นหาข้อมูลลูกค้าเฉพาะรายจากอีเมลที่ใช้สมัคร
SELECT customer_id, first_name, last_name, email, phone, address, created_at
FROM dbo.customer
WHERE email = 'demo@shop.com' OR phone = '089-999-9999';
3.3 ดูที่อยู่จัดส่งทั้งหมดของลูกค้า พร้อมระบุที่อยู่หลัก ADDRESS JOIN
แสดงรายการที่อยู่ที่ลูกค้าบันทึกไว้ในระบบ
SELECT 
    a.address_id,
    c.first_name + ' ' + c.last_name AS customer_name,
    a.recipient,
    a.phone,
    a.house_no + ' ' + a.subdistrict + ' ' + a.district + ' ' + a.province + ' ' + a.postal_code AS full_address,
    a.is_default
FROM dbo.address a
JOIN dbo.customer c ON c.customer_id = a.customer_id
ORDER BY a.customer_id, a.is_default DESC;
หมวด 4 คำสั่งค้นหาตาม Data Dictionary & Views (Orders & Order_Detail)
4.1 ค้นหาผ่าน View: Orders (ชื่อฟิลด์ตรงตาม Data Dictionary) DATA DICT VIEW
ดึงข้อมูลคำสั่งซื้อผ่าน View Orders (order_id, customer_id, order_date, total_amount, order_status)
SELECT order_id, customer_id, order_date, total_amount, order_status
FROM dbo.Orders
WHERE order_status = 'shipped';
4.2 ค้นหาผ่าน View: Order_Detail (ชื่อฟิลด์ตรงตาม Data Dictionary) DATA DICT VIEW
ดึงข้อมูลรายละเอียดคำสั่งซื้อผ่าน View Order_Detail (order_detail_id, order_id, product_id, quantity, unit_price)
SELECT order_detail_id, order_id, product_id, quantity, unit_price
FROM dbo.Order_Detail
WHERE order_id = 5;
4.3 รวม View: Orders และ Order_Detail เข้าด้วยกัน VIEW JOIN
เชื่อมโยง 2 Views เพื่อดูคำสั่งซื้อและรายการสินค้าตามเอกสารส่งอาจารย์
SELECT 
    o.order_id,
    o.order_date,
    o.order_status,
    od.order_detail_id,
    p.name AS product_name,
    od.quantity,
    od.unit_price,
    (od.quantity * od.unit_price) AS line_total
FROM dbo.Orders o
JOIN dbo.Order_Detail od ON od.order_id = o.order_id
JOIN dbo.product p ON p.product_id = od.product_id
ORDER BY o.order_id DESC;
หมวด 5 คำสั่งค้นหาข้อมูลการบริหารจัดการ (Admin, Employee, Supplier, Shipment)
5.1 ดูรายชื่อผู้ดูแลระบบ (Admin) ทั้ง 5 บัญชีกลุ่ม ADMIN LIST
แสดงข้อมูลสมาชิกกลุ่มที่เป็น Admin ทั้ง 5 คน (รหัส, ชื่อ, อีเมล, เบอร์โทร)
SELECT admin_id, name, email, phone
FROM dbo.admin
ORDER BY admin_id ASC;
5.2 ดูรายชื่อซัพพลายเออร์ พร้อมพนักงานและแอดมินที่ดูแล (JOIN 3 ตาราง) RELATIONSHIP ER
ตรวจสอบความสัมพันธ์ตาม ER Diagram ระหว่าง ซัพพลายเออร์, พนักงานผู้ติดต่อ, และแอดมิน
SELECT 
    s.supplier_id,
    s.name AS supplier_name,
    s.contact_name,
    s.phone,
    e.first_name + ' ' + e.last_name AS employee_in_charge,
    e.position AS employee_position,
    a.name AS admin_supervisor
FROM dbo.supplier s
LEFT JOIN dbo.employee e ON e.employee_id = s.employee_id
LEFT JOIN dbo.admin a ON a.admin_id = s.admin_id;
5.3 ตรวจสอบสถานะการจัดส่งพัสดุและเลขแทร็กกิ้ง (Shipment Tracking) TRACKING
ดูรายการพัสดุที่กำลังจัดส่ง เลข Tracking ผู้ให้บริการขนส่ง (Flash, Kerry) และวันที่ส่ง
SELECT 
    shipment_id,
    order_id,
    carrier,
    tracking_no,
    status,
    shipping_address,
    shipped_at
FROM dbo.shipment
ORDER BY shipment_id DESC;
หมวด 6 คำสั่งวิเคราะห์และสรุปยอดขายเชิงธุรกิจ (Analytics & Aggregation)
6.1 สรุปยอดขายรวมทั้งหมด และ ยอดขายแยกตามสถานะออเดอร์ REVENUE SUMMARY
คำนวณรายได้รวมจริง (เฉพาะออเดอร์ที่ paid, shipped, หรือ completed)
SELECT 
    status,
    COUNT(*) AS order_count,
    SUM(total_price) AS total_revenue,
    AVG(total_price) AS avg_order_value
FROM dbo.[order]
GROUP BY status;
6.2 สรุปยอดขายแยกตามหมวดหมู่สินค้า 4 ประเภท CATEGORY SALES
เปรียบเทียบยอดขายระหว่าง เสื้อยืด, เสื้อโปโล, เสื้อแขนยาว, และกางเกง
SELECT 
    p.category,
    SUM(oi.quantity) AS total_units_sold,
    SUM(oi.total_price) AS total_sales_baht
FROM dbo.order_item oi
JOIN dbo.product p ON p.product_id = oi.product_id
JOIN dbo.[order] o ON o.order_id = oi.order_id
WHERE o.status IN ('paid', 'shipped', 'completed')
GROUP BY p.category
ORDER BY total_sales_baht DESC;
6.3 หาสินค้าขายดีที่สุด 5 อันดับแรก (Top 5 Best Sellers) TOP SELLING
จัดอันดับสินค้าที่มียอดขายจำนวนชิ้นสูงสุดในร้าน
SELECT TOP 5
    p.product_id,
    p.name AS product_name,
    p.category,
    SUM(oi.quantity) AS units_sold,
    SUM(oi.total_price) AS revenue_generated
FROM dbo.order_item oi
JOIN dbo.product p ON p.product_id = oi.product_id
JOIN dbo.[order] o ON o.order_id = oi.order_id
WHERE o.status IN ('paid', 'shipped', 'completed')
GROUP BY p.product_id, p.name, p.category
ORDER BY units_sold DESC;
6.4 สรุปวิธีการชำระเงินยอดนิยม (Payment Method Popularity) PAYMENT STATS
ดูว่าลูกค้าชอบจ่ายเงินด้วยช่องทางไหนมากที่สุด (โอนผ่านธนาคาร, PromptPay, บัตร, ปลายทาง)
SELECT 
    method,
    COUNT(*) AS transaction_count,
    SUM(amount) AS total_collected
FROM dbo.payment
WHERE status = 'paid'
GROUP BY method
ORDER BY transaction_count DESC;
